simple_genetic_algorithm.simple_genetic_algorithm

Contains the implementation of a simple Genetic Algorithm framework for hyperparameter optimization.

There is the get_random helper function to create random parameter values, and the abstract base class GeneticInstance which needs to be inherited and the abstract methods implemented.

 29    """
 30    A helper function to create the get_random_param_value method.
 31
 32    Parameters:
 33    - args: Variable number of arguments to determine the range or list for random value generation.
 34

 40    else:
 41        if type(args[0]) == int:
 42            if len(args) == 1:
 43                return random.randint(0, args[0])
 44            else:
 45                return random.randint(args[0], args[1])

 47            if len(args) == 1:

 49            else:
 50                return random.uniform(args[0], args[1])
 51        elif type(args[0]) == list:
 52            return random.choice(args[0])
 53        else:
 54            return random.random()

276    def init_population(self, population_size, init_method='random', param_ranges=None, mean=0, std_dev=1, custom_params=None):
277        """
278        Initializes the population with different initialization methods.
279
280        Parameters:
281        - population_size (int): Size of the population.
282        - init_method (str): Initialization method: 'random', 'uniform', 'gaussian', or 'custom'.
283        - param_ranges (dict): Dictionary of min and max values for parameters (required for 'uniform' and 'gaussian').
284        - mean (float): Mean for Gaussian distribution (used in 'gaussian' method).
285        - std_dev (float): Standard deviation for Gaussian distribution (used in 'gaussian' method).
286        - custom_params (dict): Dictionary of custom initial values for parameters (required for 'custom').
287
288        Returns:
289        - list: List of individuals, each represented as [params, fitness].
290        """
291        new_population = []
292        
293        for _ in range(population_size):
294            params = dict()
295            
296            if init_method == 'random':
297                # Random initialization (original standard method)
298                for param in self.params:
299                    params[param] = self.get_random_param_value(param)
300
301            elif init_method == 'uniform':
302                # Uniform initialization: sample within a given range for each parameter
303                if param_ranges is None:
304                    raise ValueError("param_ranges must be provided for 'uniform' initialization.")
305                for param in self.params:
306                    min_val, max_val = param_ranges.get(param, (0, 1))
307                    params[param] = random.uniform(min_val, max_val)
308
309            elif init_method == 'gaussian':
310                # Gaussian initialization: sample from a normal distribution
311                if param_ranges is None:
312                    raise ValueError("param_ranges must be provided for 'gaussian' initialization.")
313                for param in self.params:
314                    min_val, max_val = param_ranges.get(param, (0, 1))
315                    # Ensure the generated value is within the parameter range
316                    while True:
317                        value = np.random.normal(mean, std_dev)
318                        if min_val <= value <= max_val:
319                            params[param] = value
320                            break
321
322            elif init_method == 'custom':
323                # Custom initialization: use pre-defined custom values
324                if custom_params is None:
325                    raise ValueError("custom_params must be provided for 'custom' initialization.")
326                for param in self.params:
327                    params[param] = custom_params.get(param, self.get_random_param_value(param))
328
329            else:
330                raise ValueError(f"Unknown initialization method: {init_method}")
331            
332            new_population.append([params, float('-inf')])
333        
334        return new_population

338    def select_individuals(self, population, population_size, method='elitism', k=0.2, tournament_size=3):
339        """
340        Selects individuals from the population based on the chosen selection method.
341
342        Parameters:
343        - population (list): List of individuals, each as [params, fitness].
344        - population_size (int): Total number of individuals in the population.
345        - method (str): Selection method: 'elitism', 'roulette', or 'tournament'.
346        - k (float): Fraction of individuals to select (for elitism and roulette).
347        - tournament_size (int): Number of individuals per tournament (for tournament method).
348
349        Returns:
350        - list: Selected individuals from the population.
351        """
352        population = copy.deepcopy(population)
353        
354        if method == 'elitism':    # (original standard method)
355            population = sorted(population, key=lambda x: x[1], reverse=True)
356            selected_population = population[:int(k * population_size)]
357
358        elif method == 'roulette':
359            total_fitness = sum(ind[1] for ind in population)
360            if total_fitness == 0:
361                selected_population = random.sample(population, int(k * population_size))
362            else:
363                selection_probs = [ind[1] / total_fitness for ind in population]
364                selected_population = random.choices(population, weights=selection_probs, k=int(k * population_size))
365
366        elif method == 'tournament':
367            selected_population = []
368            for _ in range(int(k * population_size)):
369                competitors = random.sample(population, tournament_size)
370                winner = max(competitors, key=lambda x: x[1])
371                selected_population.append(winner)
372
373        else:
374            raise ValueError(f"Unknown selection method: {method}")
375
376        return selected_population

380    def crossover(self, parents, possibility=0.5, crossover_method='uniform', crossover_points=(0, 1)):
381        """
382        Performs crossover to create offspring using different crossover methods.
383
384        Parameters:
385        - parents (list): List of parent individuals, each represented as [params, fitness].
386        - possibility (float, optional): Probability of crossover. Defaults to 0.5.
387        - crossover_method (str): Crossover method: 'one_point', 'two_point', or 'uniform'.
388        - crossover_points (tuple): Points used for 'one_point' or 'two_point' crossover.
389
390        Returns:
391        - list: Offspring population.
392        """
393        parents = copy.deepcopy(parents)
394        offspring_population = []
395        end_idx = len(parents)
396        last_one = None
397        if end_idx % 2 != 0:
398            end_idx -= 1
399            last_one = end_idx
400
401        for parent_idx in range(0, end_idx, 2):
402            parent_1 = parents[parent_idx]
403            parent_2 = parents[parent_idx + 1]
404
405            if random.random() > possibility:
406                if crossover_method == 'one_point':
407                    # One-Point Crossover: Select a point and swap before and after
408                    crossover_point = random.randint(0, len(parent_1[0]) - 1)
409                    child_params = {key: parent_1[0][key] if list(parent_1[0].keys()).index(key) < crossover_point else parent_2[0][key]
410                                    for key in parent_1[0]}
411                    offspring_population.append([child_params, float('-inf')])
412
413                elif crossover_method == 'two_point':
414                    # Two-Point Crossover: Select two points and swap between them
415                    point1, point2 = sorted(crossover_points)
416                    child_params = {key: parent_1[0][key] if point1 <= list(parent_1[0].keys()).index(key) < point2 else parent_2[0][key]
417                                    for key in parent_1[0]}
418                    offspring_population.append([child_params, float('-inf')])
419
420                elif crossover_method == 'uniform':
421                    # Uniform Crossover: Each parameter has an equal chance to come from either parent
422                    child_params = {key: parent_1[0][key] if random.random() < 0.5 else parent_2[0][key]
423                                    for key in parent_1[0]}
424                    offspring_population.append([child_params, float('-inf')])
425
426                else:
427                    raise ValueError(f"Unknown crossover method: {crossover_method}")
428            else:
429                offspring_population.append(parent_1)
430
431            # Second parent -> child or him/herself
432            if random.random() > possibility:
433                if crossover_method == 'one_point':
434                    crossover_point = random.randint(0, len(parent_1[0]) - 1)
435                    child_params = {key: parent_1[0][key] if list(parent_1[0].keys()).index(key) < crossover_point else parent_2[0][key]
436                                    for key in parent_1[0]}
437                    offspring_population.append([child_params, float('-inf')])
438
439                elif crossover_method == 'two_point':
440                    point1, point2 = sorted(crossover_points)
441                    child_params = {key: parent_1[0][key] if point1 <= list(parent_1[0].keys()).index(key) < point2 else parent_2[0][key]
442                                    for key in parent_1[0]}
443                    offspring_population.append([child_params, float('-inf')])
444
445                elif crossover_method == 'uniform':
446                    child_params = {key: parent_1[0][key] if random.random() < 0.5 else parent_2[0][key]
447                                    for key in parent_1[0]}
448                    offspring_population.append([child_params, float('-inf')])
449
450            else:
451                offspring_population.append(parent_2)
452
453        # Add parent without a pair if population size is odd
454        if last_one is not None:
455            offspring_population.append(parents[last_one])
456
457        return offspring_population

461    def mutate_population(self, population, mutation_rate, population_size, 
462                          mutation_method='random', 
463                          gaussian_std_dev=None, 
464                          uniform_min=None, uniform_max=None, 
465                          discrete_choices=None):
466        """
467        Mutates the population based on the mutation rate using different mutation methods.
468
469        Parameters:
470        - population (list): List of individuals, each represented as [params, fitness].
471        - mutation_rate (float): Probability of mutation for an individual's hyperparameters.
472        - population_size (int): Amount of individuals/solution-fitness pairs.
473        - mutation_method (str): Mutation method: 'random', 'gaussian', 'uniform', or 'discrete'.
474        - gaussian_std_dev (float, optional): Standard deviation for gaussian mutation.
475        - uniform_min (float, optional): Minimum value for uniform mutation.
476        - uniform_max (float, optional): Maximum value for uniform mutation.
477        - discrete_choices (list, optional): List of discrete choices for mutation.
478
479        Returns:
480        - list: Mutated population.
481        """
482        population = copy.deepcopy(population)
483        
484        for random_individual_idx in range(len(population)):
485            if random.random() <= mutation_rate:
486                for param in self.params:
487                    if mutation_method == 'random':
488                        # Random Mutation: Replace with random value from get_random_param_value
489                        if random.random() > 0.5:
490                            population[random_individual_idx][0][param] = self.get_random_param_value(param)
491
492                    elif mutation_method == 'gaussian':
493                        # Gaussian Mutation: Add Gaussian noise to the parameter value
494                        if gaussian_std_dev is None:
495                            raise ValueError("Missing 'gaussian_std_dev' for gaussian mutation")
496                        noise = random.gauss(0, gaussian_std_dev)
497                        population[random_individual_idx][0][param] += noise
498
499                    elif mutation_method == 'uniform':
500                        # Uniform Mutation: Replace the parameter with a random value within a specific range
501                        if uniform_min is None or uniform_max is None:
502                            raise ValueError("Missing 'uniform_min' or 'uniform_max' for uniform mutation")
503                        population[random_individual_idx][0][param] = random.uniform(uniform_min, uniform_max)
504
505                    elif mutation_method == 'discrete':
506                        # Discrete Mutation: Change the parameter by picking a random discrete value
507                        if discrete_choices is None:
508                            raise ValueError("Missing 'discrete_choices' for discrete mutation")
509                        population[random_individual_idx][0][param] = random.choice(discrete_choices)
510
511                    else:
512                        raise ValueError(f"Unknown mutation method: {mutation_method}")
513
514                # Reset fitness after mutation
515                population[random_individual_idx][1] = float('-inf')
516
517        return population

399            end_idx -= 1
400            last_one = end_idx

403            parent_1 = parents[parent_idx]
404            parent_2 = parents[parent_idx + 1]
405

409                    crossover_point = random.randint(0, len(parent_1[0]) - 1)

416                    point1, point2 = sorted(crossover_points)

418                                    for key in parent_1[0]}
